# Day One - Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A hosted Next.js web app where an AI Builder candidate discovers a problem by interviewing 4 conflicting LLM personas (with hidden facts + a 20-question budget), synthesizes it, builds a solution spec with an AI build agent + component library, defends it in a boardroom group chat, and gets an evidence-cited LLM-judge scorecard with human-override controls.

**Architecture:** Next.js App Router (single app, no DB). All session state in a server-side in-memory store; every action appends to one append-only trace (`TraceEvent[]`) which is the single source of truth for the judge. Personas reveal hidden facts via an invisible marker protocol (`⟦HF-n⟧`) the server strips and logs - making the hidden-facts audit fully deterministic. Judge = 6 parallel Claude calls (one per rubric dimension, structured output) + deterministic audit + weighted assembly with governance gate.

**Tech Stack:** Next.js 15 (App Router, TypeScript), Tailwind, `@anthropic-ai/sdk` (model `claude-opus-4-8`, adaptive thinking, `zodOutputFormat` for judge), Vitest for core-logic tests, Vercel hosting.

**Spec:** `docs/superpowers/specs/2026-06-10-proving-ground-design.md`
**Persona ground truth:** `docs/persona-dossiers.md` (all persona content MUST come from here verbatim - research-cited)

**Timebox note:** This is a 3-hour-cap case study. TDD applies to deterministic core logic (store, budget, marker extraction, scoring assembly). LLM endpoints and UI are verified by running them (manual + curl), not unit tests - testing LLM output deterministically is not possible and the brief rewards scoping judgment.

```ts
import { Session, TraceEvent, PersonaId, Phase } from "./types";

// globalThis survives Next.js dev hot-reload; acceptable for a prototype (spec §8: no DB in v1)
const g = globalThis as unknown as { __pgSessions?: Map<string, Session> };
const sessions = (g.__pgSessions ??= new Map<string, Session>());

export function createSession(id?: string): Session {
  const s: Session = {
    id: id ?? `pg-${Math.random().toString(36).slice(2, 10)}`,
    createdAt: Date.now(),
    questionBudget: 20,
    questionsUsed: 0,
    questionsPerPersona: { partner: 0, senior: 0, it: 0, risk: 0 },
    revealedFacts: [],
    trace: [],
    status: "active",
  };
  sessions.set(s.id, s);
  return s;
}

export function getSession(id: string): Session | undefined {
  return sessions.get(id);
}

export function appendEvent(
  sessionId: string,
  ev: Omit<TraceEvent, "id" | "ts">
): TraceEvent {
  const s = sessions.get(sessionId);
  if (!s) throw new Error(`unknown session ${sessionId}`);
  const event: TraceEvent = {
    ...ev,
    id: `ev-${String(s.trace.length + 1).padStart(4, "0")}`,
    ts: Date.now(),
  };
  s.trace.push(event);
  return event;
}

export function spendQuestion(sessionId: string, persona: PersonaId): boolean {
  const s = sessions.get(sessionId);
  if (!s) throw new Error(`unknown session ${sessionId}`);
  if (s.questionsUsed >= s.questionBudget) return false;
  s.questionsUsed++;
  s.questionsPerPersona[persona]++;
  return true;
}

export function recordFacts(sessionId: string, factIds: string[], phase: Phase): void {
  const s = sessions.get(sessionId);
  if (!s) throw new Error(`unknown session ${sessionId}`);
  for (const f of factIds) {
    if (s.revealedFacts.includes(f)) continue;
    s.revealedFacts.push(f);
    appendEvent(sessionId, {
      phase,
      actor: "system",
      type: "fact_revealed",
      content: `Hidden fact surfaced: ${f}`,
      refs: [f],
    });
  }
}

export function loadSession(s: Session): void {
  sessions.set(s.id, s); // for seeded sample runs
}
```

```ts
import { HIDDEN_FACTS_LEDGER } from "./scenario";

export function extractMarkers(text: string): { cleaned: string; factIds: string[] } {
  const factIds: string[] = [];
  const cleaned = text
    .replace(/⟦(HF-\d+)⟧/g, (_, id: string) => {
      if (HIDDEN_FACTS_LEDGER[id]) factIds.push(id);
      return "";
    })
    .trim();
  return { cleaned, factIds };
}
```

`app/api/chat/route.ts`:

```ts
import { NextRequest, NextResponse } from "next/server";
import { anthropic, MODEL } from "@/lib/anthropic";
import { getSession, appendEvent, spendQuestion, recordFacts } from "@/lib/store";
import { PERSONAS, buildPersonaSystemPrompt } from "@/lib/personas";
import { extractMarkers } from "@/lib/markers";
import { PersonaId } from "@/lib/types";

const PANEL_SYSTEM = `You are an AI assistant available to a candidate working through a simulated
consulting engagement (discovering where AI agents could help an audit practice). Help them however
they ask - brainstorm questions, critique framing, draft text, explain audit concepts. Be genuinely
useful and concise. Do not refuse to do their work for them; how they use you is part of what is
being assessed, not your concern.`;

export async function POST(req: NextRequest) {
  const { sessionId, target, message } = (await req.json()) as {
    sessionId: string;
    target: PersonaId | "panel";
    message: string;
  };
  const session = getSession(sessionId);
  if (!session || session.status !== "active")
    return NextResponse.json({ error: "no active session" }, { status: 404 });

  if (target === "panel") {
    appendEvent(sessionId, { phase: "discovery", actor: "candidate", type: "panel_query", content: message });
    const history = session.trace
      .filter(e => e.type === "panel_query" || e.type === "panel_reply")
      .map(e => ({
        role: e.type === "panel_query" ? ("user" as const) : ("assistant" as const),
        content: e.content,
      }));
    const resp = await anthropic.messages.create({
      model: MODEL,
      max_tokens: 1024,
      system: PANEL_SYSTEM,
      output_config: { effort: "low" },
      messages: history,
    });
    const text = resp.content.find(b => b.type === "text")?.text ?? "";
    appendEvent(sessionId, { phase: "discovery", actor: "panel", type: "panel_reply", content: text });
    return NextResponse.json({ reply: text, questionsLeft: session.questionBudget - session.questionsUsed });
  }

  // Persona chat - costs a question
  if (!spendQuestion(sessionId, target))
    return NextResponse.json({ error: "question budget exhausted" }, { status: 429 });

  appendEvent(sessionId, { phase: "discovery", actor: "candidate", type: "candidate_message", content: `[to ${target}] ${message}` });

  const persona = PERSONAS[target];
  const history = session.trace
    .filter(
      e =>
        (e.type === "candidate_message" && e.content.startsWith(`[to ${target}]`)) ||
        (e.type === "persona_message" && e.actor === `persona:${target}`)
    )
    .map(e => ({
      role: e.type === "candidate_message" ? ("user" as const) : ("assistant" as const),
      content: e.type === "candidate_message" ? e.content.replace(`[to ${target}] `, "") : e.content,
    }));

  const resp = await anthropic.messages.create({
    model: MODEL,
    max_tokens: 1024,
    system: [{ type: "text", text: buildPersonaSystemPrompt(persona), cache_control: { type: "ephemeral" } }],
    output_config: { effort: "low" },
    messages: history,
  });

  const raw = resp.content.find(b => b.type === "text")?.text ?? "";
  const { cleaned, factIds } = extractMarkers(raw);
  appendEvent(sessionId, { phase: "discovery", actor: `persona:${target}`, type: "persona_message", content: cleaned });
  if (factIds.length) recordFacts(sessionId, factIds, "discovery");

  return NextResponse.json({
    reply: cleaned,
    questionsLeft: session.questionBudget - session.questionsUsed,
  });
}
```

```ts
conflicting priorities - use them):

${Object.values(PERSONAS).map(p => `=== ${p.id.toUpperCase()} ===\n${buildPersonaSystemPrompt(p)}`).join("\n\n")}

now reveal one pointedly, with its ⟦HF-n⟧ marker, if the proposal stumbles into it). 1-3 sentences
each, in their distinct voices. Not every persona must speak every round - only those with something
real to say (minimum 2).`;

  const resp = await anthropic.messages.parse({
    model: MODEL,
    max_tokens: 2048,
    thinking: { type: "adaptive" },
    system,
    messages: [{ role: "user", content: `Boardroom so far:\n${priorBoardroom}\n\nGenerate the next round of reactions.` }],
    output_config: { format: zodOutputFormat(BoardroomRound) },
  });

  const round = resp.parsed_output!;
  for (const r of round.reactions) {
    // markers in boardroom messages also count as fact reveals

    const { recordFacts } = await import("@/lib/store");

    r.message = cleaned;
  }
  return NextResponse.json(round);
}
```

```ts
import { z } from "zod";
import { zodOutputFormat } from "@anthropic-ai/sdk/helpers/zod";
import { anthropic, MODEL } from "./anthropic";
import { getSession } from "./store";
import { DIMENSIONS, PROFILES } from "./rubric";
import { CRITICAL_FACTS, HIDDEN_FACTS_LEDGER } from "./scenario";
import { DimensionScore, Scorecard, Session } from "./types";

const JudgeOutput = z.object({
  score: z.number().int().min(1).max(4),
  confidence: z.enum(["low", "medium", "high"]),
  evidence: z.array(z.object({ quote: z.string(), eventId: z.string() })).min(1).max(5),
  counterEvidence: z.string(),
  rationale: z.string(),
});

const GateOutput = z.object({
  triggered: z.boolean(),
  reason: z.string(),
});

export function runAudit(sessionId: string): Scorecard["audit"] {
  const s = getSession(sessionId)!;
  return {
    criticalFactsFound: s.revealedFacts.filter(f => CRITICAL_FACTS.includes(f)),
    usefulFactsFound: s.revealedFacts.filter(f => !CRITICAL_FACTS.includes(f)),
    questionsUsed: s.questionsUsed,
    questionsPerPersona: s.questionsPerPersona,
    componentsReused: s.trace.filter(e => e.type === "component_used").length,
    revisedAfterBoardroom: s.trace.some(e => e.type === "revision_saved"),
  };
}

export function assembleScorecard(
  s: Session,
  scores: DimensionScore[],
  profile: keyof typeof PROFILES,
  gateReason: string | null
): Scorecard {
  const weights = PROFILES[profile];
  let weightedTotal = scores.reduce(
    (acc, d) => acc + d.score * (weights[d.dimension as keyof typeof weights] ?? 0),
    0
  );
  if (gateReason) weightedTotal = Math.min(weightedTotal, 2.0); // gate caps regardless of dimensions
  return { dimensions: scores, audit: runAudit(s.id), gateTriggered: gateReason, weightedTotal, profile };
}

export async function judgeSession(sessionId: string, profile: keyof typeof PROFILES): Promise<Scorecard> {
  const s = getSession(sessionId)!;
  const audit = runAudit(sessionId);
  const auditText = `DETERMINISTIC AUDIT (ground truth, not LLM opinion):
Critical hidden facts surfaced: ${audit.criticalFactsFound.join(", ") || "NONE"} of ${CRITICAL_FACTS.join(", ")}
(${CRITICAL_FACTS.map(f => `${f}: ${HIDDEN_FACTS_LEDGER[f].summary}`).join(" | ")})
Useful facts surfaced: ${audit.usefulFactsFound.join(", ") || "none"}
Questions used: ${audit.questionsUsed}/20 - per persona: ${JSON.stringify(audit.questionsPerPersona)}
Components reused: ${audit.componentsReused}. Revised after boardroom: ${audit.revisedAfterBoardroom}`;

  // One judge call per dimension, in parallel; each sees only its evidence slice (spec §4)
  const dimScores = await Promise.all(
    DIMENSIONS.map(async (dim): Promise<DimensionScore> => {
      const slice = dim.slice(s)
        .map(e => `[${e.id}] (${e.actor}/${e.type}): ${e.content.slice(0, 1500)}`)
        .join("\n");
      const resp = await anthropic.messages.parse({
        model: MODEL,
        max_tokens: 2048,
        thinking: { type: "adaptive" },
        system: `You are an evaluation judge for an AI Builder hiring simulation. You score ONE
dimension only, strictly from the trace evidence provided. Every claim in your rationale must cite
event IDs from the trace. You never see the candidate's name or background. If evidence is thin,
say so via low confidence - do not invent. Genuinely consider counter-evidence before scoring.`,
        messages: [{
          role: "user",
          content: `DIMENSION: ${dim.id} - ${dim.name}\n${dim.prompt}\n\n${auditText}\n\nTRACE EVIDENCE:\n${slice || "(no events - score 1 with high confidence)"}`,
        }],
        output_config: { format: zodOutputFormat(JudgeOutput) },
      });
      const out = resp.parsed_output!;
      return { dimension: dim.id, score: out.score as 1 | 2 | 3 | 4, confidence: out.confidence, evidence: out.evidence, counterEvidence: out.counterEvidence, rationale: out.rationale };
    })
  );

  // Governance gate check (spec §3): one focused call over the full trace summary
  const gateResp = await anthropic.messages.parse({
    model: MODEL,
    max_tokens: 1024,
    thinking: { type: "adaptive" },
    system: `You check for disqualifying red flags in a hiring-simulation trace. Trigger ONLY on
clear evidence of: (a) fabricating stakeholder facts that contradict the trace, (b) proposing fully
autonomous agents in regulated sign-off paths AFTER the Risk stakeholder flagged the constraint,
(c) deliverables wholesale AI-generated with no defense under boardroom questioning, (d) gaming the
simulation rather than engaging. Honest mistakes and weak answers are NOT red flags. Default to
triggered=false when uncertain.`,
    messages: [{
      role: "user",
      content: `${auditText}\n\nSYNTHESIS: ${JSON.stringify(s.synthesis)}\nSPEC: ${JSON.stringify(s.spec)}\nBOARDROOM: ${s.trace.filter(e => e.type === "boardroom_message").map(e => `[${e.actor}] ${e.content}`).join("\n")}\nPANEL USAGE: ${s.trace.filter(e => e.type === "panel_query").map(e => e.content).join("\n").slice(0, 3000)}`,
    }],
    output_config: { format: zodOutputFormat(GateOutput) },
  });
  const gate = gateResp.parsed_output!;

  const card = assembleScorecard(s, dimScores, profile, gate.triggered ? gate.reason : null);
  s.scorecard = card;
  s.status = "scored";
  return card;
}
```
